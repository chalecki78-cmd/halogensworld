        function toggleLanguage() {
            const newLang = currentLang === 'pl' ? 'en' : 'pl';
            applyLanguage(newLang);
        }

        function showNotify(title, message, isError = false) {
            const box = document.getElementById('system-notify');
            document.getElementById('notify-title').innerText = title;
            document.getElementById('notify-body').innerText = message;
            box.classList.toggle('error', isError);

            box.classList.remove('show');
            void box.offsetWidth;
            box.classList.add('show');

            clearTimeout(notifyTimeout);
            notifyTimeout = setTimeout(() => {
                box.classList.remove('show');
            }, 3200);
        }

        function openModal(mode) {
            document.getElementById('auth-modal').style.display = 'flex';
            document.getElementById('login-form').style.display = 'none';
            document.getElementById('register-form').style.display = 'none';
            document.getElementById('help-form').style.display = 'none';

            if (mode === 'login') switchToLogin();
            else if (mode === 'register') switchToRegister();
            else if (mode === 'help') document.getElementById('help-form').style.display = 'block';
        }

        function closeModal() {
            document.getElementById('auth-modal').style.display = 'none';
        }

        function handleModalOutsideClick(event) {
            if (event.target.id === 'auth-modal') closeModal();
        }

        function switchToRegister() {
            document.getElementById('login-form').style.display = 'none';
            document.getElementById('help-form').style.display = 'none';
            document.getElementById('register-form').style.display = 'block';
        }

        function switchToLogin() {
            document.getElementById('register-form').style.display = 'none';
            document.getElementById('help-form').style.display = 'none';
            document.getElementById('login-form').style.display = 'block';
        }

        function handleLogin() {
            const user = document.getElementById('login-user').value.trim();
            const pass = document.getElementById('login-pass').value;
            const errorLabel = document.getElementById('login-error');
            const t = translations[currentLang];

            if (!user || !pass) {
                errorLabel.innerText = t.errEmpty;
                return;
            }
            errorLabel.innerText = '';

            document.getElementById('user-role-label').innerText = 'USER: ' + user.toUpperCase();
            document.getElementById('login-trigger-btn').style.display = 'none';
            document.getElementById('logout-btn').style.display = 'inline-block';
            closeModal();
            showNotify(t.notifyGrantTitle, t.notifyGrantBody + user.toUpperCase());

            document.getElementById('login-user').value = '';
            document.getElementById('login-pass').value = '';
        }

        function handleLogout() {
            const t = translations[currentLang];
            document.getElementById('user-role-label').innerText = 'GUEST (LIMITED)';
            document.getElementById('login-trigger-btn').style.display = 'inline-block';
            document.getElementById('logout-btn').style.display = 'none';
            showNotify(t.notifyEndTitle, t.notifyEndBody);
        }

        function handleRegister() {
            const user = document.getElementById('reg-user').value.trim();
            const pass = document.getElementById('reg-pass').value;
            const errorLabel = document.getElementById('reg-error');
            const t = translations[currentLang];

            if (!user || !pass) {
                errorLabel.innerText = t.errEmpty;
                return;
            }
            errorLabel.innerText = '';

            showNotify(t.notifyCreatedTitle, user.toUpperCase() + t.notifyCreatedBody);
            switchToLogin();
            document.getElementById('login-user').value = user;
            document.getElementById('reg-user').value = '';
            document.getElementById('reg-pass').value = '';
        }

        window.onload = () => {
            applyLanguage(currentLang);

            if (screen.orientation && screen.orientation.lock) {
                screen.orientation.lock('landscape').catch(() => {});
            }
        };
    </script>
